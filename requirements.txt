# Trimmed to what the deployed app (app.py + pages/) actually imports.
# The previous version of this file was a raw `pip freeze` of the whole local
# dev environment (jupyter/jupyterlab/ipykernel/debugpy and friends included),
# plus a CUDA-specific torch build (`torch==2.13.0+cu130`) that only exists
# for Linux boxes with an NVIDIA GPU + matching driver — Streamlit Community
# Cloud's containers are CPU-only, so pip could never resolve/install that
# wheel there, which is what "Error during processing dependencies" was
# almost certainly reporting. It also pinned tensorflow and its entire
# dependency tree (keras, h5py, grpcio, ml_dtypes, flatbuffers, ...) even
# though nothing the deployed app imports uses TensorFlow — that's only used
# by train_baseline_classifier.py (an offline training script, run locally,
# never by `streamlit run app.py`); the app just reads its already-computed
# baseline_stats.json/baseline_history.json as plain JSON.
#
# Versions below are left unpinned except where the app's own code depends
# on specific behavior, so pip can resolve whatever wheel actually exists for
# Streamlit Cloud's Python version instead of chasing an exact version that
# may not have one.
streamlit
pillow
numpy
pandas
ultralytics==8.4.127
# ultralytics pulls in plain `opencv-python`, which needs GUI/X11 libraries
# (libGL, libSM, libXext, libXrender, ...) that a headless server like
# Streamlit Cloud does not have and packages.txt cannot cheaply supply. Listing
# opencv-python-headless explicitly (same `cv2` module, none of the GUI deps)
# is the standard fix for ultralytics on a server deployment.
opencv-python-headless
streamlit-drawable-canvas==0.9.3  # image-detail-view box editor (annotate_helpers.py); needs the numpy shim in that file under numpy>=1.24
